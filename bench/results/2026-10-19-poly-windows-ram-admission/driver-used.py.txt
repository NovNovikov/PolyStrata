import ctypes, json, os, subprocess, sys, time
from pathlib import Path
import psutil
ROOT = Path(__file__).resolve().parents[1]
os.chdir(ROOT)
OUT = ROOT / 'bench/results/2026-10-19-poly-windows-ram-admission'
MIB = 1048576
child_code = r'''
import ctypes, os, sys
k = ctypes.WinDLL('kernel32', use_last_error=True)
k.VirtualAlloc.argtypes = [ctypes.c_void_p, ctypes.c_size_t, ctypes.c_ulong, ctypes.c_ulong]
k.VirtualAlloc.restype = ctypes.c_void_p
k.VirtualFree.argtypes = [ctypes.c_void_p, ctypes.c_size_t, ctypes.c_ulong]
n = 8192 * 1048576
p = k.VirtualAlloc(None, n, 0x3000, 4)
if not p: raise ctypes.WinError(ctypes.get_last_error())
try:
    block = os.urandom(65536)
    for off in range(0,n,len(block)): ctypes.memmove(p+off,block,len(block))
    print('READY', flush=True)
    for line in sys.stdin:
        if line.strip() == 'QUIT': break
        if line.strip() == 'TOUCH':
            checksum = 0
            for off in range(0,n,4096): checksum ^= ctypes.c_ubyte.from_address(p+off).value
            print('TOUCHED', checksum, flush=True)
finally:
    assert k.VirtualFree(p,0,0x8000)
'''
summary = {'pressure_mib':8192,'synthetic_tokens':10000,'generate':128,'runs':[]}
holder = subprocess.Popen([sys.executable,'-u','-c',child_code],stdin=subprocess.PIPE,stdout=subprocess.PIPE,stderr=subprocess.PIPE,text=True,creationflags=subprocess.CREATE_NO_WINDOW)
try:
    assert holder.stdout.readline().strip() == 'READY'
    h = psutil.Process(holder.pid)
    common = [sys.executable,'tools/bench_poly_no_disk.py','--model',r'L:\AI_pictures_generate\text-generation-webui2\user_data\models\DeepSeek-V4-Flash-0731-Q3_K_M-custom.gguf','--tokenizer',r'L:\AI_pictures_generate\PolyStrata-data\packs\dsv4-q3_k_m\tokenizer','--profile','bench/results/2026-10-19-poly-compact-scratch/final-128k-50k/no-disk-1/expert-profile.txt','--context','131072','--tokens','10000','--batch','4096','--read-chunk','32768','--stage-mib','2848','--reserve-mib','256','--load-margin-mib','256','--threads','12','--generate','128','--requests','1','--adaptive-experts','--verify-stage-cache','--rows-vram','--no-disk-runs','1','--skip-default','--engine-arg=--compact-scratch','--engine-arg=--lookup','--engine-arg=0']
    for label,exe in [('before','build-poly-cuda12/strata-poly-before-ram-admission.exe'),('after','build-poly-cuda12/strata-poly.exe')]:
        holder.stdin.write('TOUCH\n'); holder.stdin.flush()
        assert holder.stdout.readline().startswith('TOUCHED')
        record = {'label':label,'available_before_mib':psutil.virtual_memory().available/MIB,'holder_rss_before_mib':h.memory_info().rss/MIB}
        cmd=common+['--exe',exe,'--output',str(OUT/label)]
        record['command']=cmd
        print(label,'starting',json.dumps({k:v for k,v in record.items() if k!='command'}),flush=True)
        with (OUT/f'{label}-runner.txt').open('w',encoding='utf8') as log:
            p=subprocess.run(cmd,stdout=log,stderr=subprocess.STDOUT,timeout=1200,creationflags=subprocess.CREATE_NO_WINDOW)
        record.update(exit_code=p.returncode,available_after_mib=psutil.virtual_memory().available/MIB,holder_rss_after_mib=h.memory_info().rss/MIB)
        summary['runs'].append(record)
        (OUT/'reclaim-summary.json').write_text(json.dumps(summary,indent=2),encoding='utf8')
        print(label,'finished',p.returncode,'holder RSS MiB',record['holder_rss_after_mib'],flush=True)
finally:
    try:
        holder.stdin.write('QUIT\n'); holder.stdin.flush(); holder.wait(timeout=30)
    except Exception:
        holder.kill(); holder.wait(timeout=30)
    summary['holder_exit_code']=holder.returncode
    summary['available_after_cleanup_mib']=psutil.virtual_memory().available/MIB
    (OUT/'reclaim-summary.json').write_text(json.dumps(summary,indent=2),encoding='utf8')
    print('pressure released',summary['holder_exit_code'],flush=True)
